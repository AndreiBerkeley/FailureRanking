from datetime import datetime

class MultilingualDiaryEntry:
    # Supported languages
    LOCALES = {'en', 'fr', 'es'}
    
    def __init__(self, title, entry_text, mood, weather, language='en'):
        # Data Validation
        if not all(isinstance(i, str) and len(i.strip()) > 0 for i in [title, entry_text, mood, weather]):
            raise ValueError("All inputs (title, entry, mood, weather) must be non-empty strings.")
        if language not in self.LOCALES:
            raise ValueError(f"Language must be one of {self.LOCALES}")

        self.title = title
        self.entry_text = entry_text
        self.mood = mood
        self.weather = weather
        self.timestamp = datetime.now()
        self.language = language
        self.labels = {
            'en': {'date': 'Date', 'mood': 'Mood'},
            'fr': {'date': 'Date', 'mood': 'Humeur'},
            'es': {'date': 'Fecha', 'mood': 'Estado de ánimo'}
        }

    def get_formatted_time(self, use_24h=True):
        """Returns the timestamp formatted for 12 or 24 hour display."""
        fmt = "%Y-%m-%d %H:%M:%S" if use_24h else "%Y-%m-%d %I:%M:%S %p"
        return self.timestamp.strftime(fmt)

    def analyze_sentiment(self):
        """
        Analyzes the mood/entry text. 
        In a production environment, this would call an external API (e.g., TextBlob/Google NLP).
        """
        try:
            # Simulated API call logic
            sentiment_map = {"happy": 0.9, "sad": -0.5, "neutral": 0.0}
            score = sentiment_map.get(self.mood.lower(), 0.0)
            return {"status": "success", "score": score}
        except Exception as e:
            return {"status": "error", "message": str(e)}

    def display_entry(self, use_24h=True):
        """Returns a localized summary of the diary entry."""
        l = self.labels[self.language]
        return (f"{l['date']}: {self.get_formatted_time(use_24h)}\n"
                f"Title: {self.title}\n{l['mood']}: {self.mood}")
